Mazi Coffee - Secure Data Flow

        graph LR
            %% Styling
            classDef client fill:#e3f2fd,stroke:#2196f3,stroke-width:2px;
            classDef server fill:#e8f5e9,stroke:#4caf50,stroke-width:2px;
            classDef data fill:#fff3e0,stroke:#ff9800,stroke-dasharray: 5 5;
            classDef secret fill:#ffebee,stroke:#f44336;

            subgraph Browser["🌐 CLIENT (Browser)"]
                Card["💳 Card Data"]:::data
                AES["🔑 AES Key"]:::secret
                RSA_PUB["🔓 RSA Public Key"]
                
                Card -->|1. Locked with AES| EncData["📦 Encrypted Data"]
                AES -->|2. Locked with RSA| EncKey["🔐 Encrypted Key"]
            end

            EncData -.->|3. Internet Transmission| Server
            EncKey -.->|3. Internet Transmission| Server

            subgraph ServerSide["🔒 SERVER (Secure)"]
                Server["⚙️ Backend Processing"]:::server
                RSA_PRIV["🗝️ RSA Private Key"]:::secret
                
                RSA_PRIV -->|4. Unlocks| EncKey
                EncKey -->|Retrieves| AES_Recovered["🔑 AES Key"]
                AES_Recovered -->|5. Unlocks| EncData
                EncData -->|Retrieves| RealData["💳 Card Data"]:::data
            end

            class Browser client;
            class ServerSide server;
